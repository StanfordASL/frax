    Inverse Kinematics

    Differential IK

    min sum_tasks || weight * (J v - v_des) ||^2 + (reg + lam) * ||v||^2 s.t. A v == b; G v <= h

    i.e. - Minimimize the weighted task velocity error, plus regularization and damping terms for stability - Subject to a set of equality and inequality constraints

    Note that since we use ElastiQP as our solver, inequality constraints are automatically (and efficiently) relaxed with an L1 penalty term. These constraints are held exactly when the problem is feasible, and only relaxed when the problem is infeasible*

    *See the ElastiQP paper for more details, and more precise conditions on relaxation and when this occurs

    Equality constraints are still held as hard constraints with ElastiQP, unless explicitly posed as an elastic equality. In this case, the equality is converted to a pair of opposing elastic inequalities.

class DiffIKSolver:
    """Differential IK with ElastiQP

    Args:
        robot (Robot): Robot model
        tasks (Sequence[Task]): Tracking tasks
        constraints (Sequence[InequalityConstraint | EqualityConstraint]):
            Constraints. Defaults to an empty tuple (unconstrained)
        regularization (ArrayLike): Tikhonov regularization on v,
            scalar or shape (nv,). Defaults to 1e-8
        damping (float): Levenberg-Marquardt damping, per unit of task cost.
            Defaults to 0.0 (no damping)
        solver_tol (float): QP solver tolerance. Defaults to 1e-5
        warm_start (bool): Whether to warm-start each QP from the previous
            solution. Defaults to True.
    """

    def __init__(
        self,
        robot: Robot,
        tasks: Sequence[Task],
        constraints: Sequence[InequalityConstraint | EqualityConstraint] = (),
        regularization: ArrayLike = 1e-8,
        damping: float = 0.0,
        solver_tol: float = 1e-5,
        warm_start: bool = True,
    ):
        assert len(tasks) > 0
        assert damping >= 0
        self.robot = robot
        self.tasks = tuple(tasks)
        self.constraints = tuple(constraints)
        self.regularization = np.broadcast_to(
            np.asarray(regularization, dtype=float), (robot.nv,)
        ).copy()
        self.damping = damping
        self.solver_tol = solver_tol
        self.warm_start = warm_start
        self.max_gain = _max_gain(self.tasks + self.constraints)
        for c in self.constraints:
            assert isinstance(c, InequalityConstraint | EqualityConstraint), (
                f"{type(c).__name__} is neither an InequalityConstraint nor an "
                + "EqualityConstraint"
            )
        for term in self.tasks + self.constraints:
            assert term.robot is robot, (
                f"{type(term).__name__} was built for a different robot than the solver's"
            )
        self.inequalities = tuple(
            c for c in self.constraints if isinstance(c, InequalityConstraint)
        )
        self.equalities = tuple(
            c for c in self.constraints if isinstance(c, EqualityConstraint)
        )
        self.n_eq = sum(c.num_rows for c in self.equalities)

        def penalties(c):
            return np.broadcast_to(np.asarray(c.penalty, dtype=float), (c.num_rows,))

        self.penalty = np.concatenate(
            [penalties(c) for c in self.inequalities] + [np.zeros(0)]
        )
        self.n_ineq = len(self.penalty)

    def init(self, q0: Array) -> SolverState:
        nv = self.robot.nv
        qp_warm_start = (jnp.zeros(nv), jnp.zeros(self.n_eq), jnp.zeros(self.n_ineq))
        q = jnp.asarray(q0, dtype=jnp.float64)  # float64 for elastiqp
        return SolverState(q=q, qp_warm_start=qp_warm_start)

    def _qp(self, q: Array, ref: Any):
        """The diff IK QP at q, and the cost of q

        Returns:
            tuple: P, c, G, h, A, b (A and b are None if there are no hard
                equalities), and the cost of q (the QP objective at v = 0)
        """
        robot = self.robot
        assert q.shape == (robot.nq,), f"q has shape {q.shape}, expected ({robot.nq},)"
        Js, v_des, ws = zip(*(task.linearize(q, ref) for task in self.tasks))
        w = jnp.concatenate(ws)
        WJ = w[:, None] * jnp.concatenate(Js)
        w_v_des = w * jnp.concatenate(v_des)
        task_error = jnp.sum(w_v_des**2)
        P = WJ.T @ WJ + jnp.diag(self.regularization + self.damping * task_error)
        c = -w_v_des @ WJ

        Gs, hs = [jnp.zeros((0, robot.nv))], [jnp.zeros(0)]
        for con in self.inequalities:
            G, h = con.rows(q, ref)
            Gs.append(G)
            hs.append(h)
        G = jnp.concatenate(Gs)
        h = jnp.concatenate(hs)
        cost = task_error + jnp.maximum(-h, 0.0) @ self.penalty

        A = b = None
        if self.equalities:
            As, bs = zip(*(con.rows(q, ref) for con in self.equalities))
            A = jnp.concatenate(As)
            b = jnp.concatenate(bs)
            # Hard equalities have no penalty, so their residual |A 0 - b| is
            # weighted like a default elastic row's
            cost += DEFAULT_PENALTY * jnp.sum(jnp.abs(b))
        return P, c, G, h, A, b, cost

    def cost(self, q: Array, ref: Any) -> Array:
        """The cost of q: the QP objective at v = 0 (if the robot doesn't move),
        for the given reference data"""
        *_, cost = self._qp(q, ref)
        return cost

    def solve(
        self, q: Array, ref: Any, qp_warm_start: tuple[Array, Array, Array]
    ) -> tuple[Array, tuple[Array, Array, Array], SolverInfo]:
        """Solve the diff IK QP

        Args:
            q (Array): Configuration, shape (nq,)
            ref (Any): Per-step reference data (any pytree)
            qp_warm_start (tuple[Array, Array, Array]): QP warm start (x, y, z)

        Returns:
            v (Array): Generalized velocity, shape (nv,)
            qp_warm_start (tuple[Array, Array, Array]): QP solution (x, y, z),
                to warm-start the next QP
            info (SolverInfo): Solver info, with the cost of q
        """
        P, c, G, h, A, b, cost = self._qp(q, ref)
        result = elastiqp.jax.solve(
            P,
            c,
            G,
            h,
            penalty=self.penalty,
            A=A,
            b=b,
            eps_abs=self.solver_tol,
            warm_start=qp_warm_start if self.warm_start else None,
        )
        info = SolverInfo(result.iters, result.converged, cost)
        return result.x, (result.x, result.y, result.z), info

    def step(
        self, state: SolverState, ref: Any, dt: float
    ) -> tuple[SolverState, Array, SolverInfo]:
        """One diff IK step

        Args:
            state (SolverState): Current state
            ref (Any): Per-step reference data (any pytree)
            dt (float): Timestep

        Returns:
            state (SolverState): New state
            v (Array): Generalized velocity, shape (nv,)
            info (SolverInfo): Solver info
        """
        v, qp_warm_start, info = self.solve(state.q, ref, state.qp_warm_start)
        q = self.robot.integrate(state.q, v, dt)
        return SolverState(q=q, qp_warm_start=qp_warm_start), v, info

    def converge(
        self, state: SolverState, ref: Any, max_iters: int, dt_max: float, tol: float
    ) -> tuple[SolverState, Array]:
        """Run differential IK until convergence

        Note: This method should generally only be called as a component of a
        global IK pipeline, or at initialization of a solver. During online
        operation, or sequential calls in an offline pipeline, generally
        only a single diff IK step() should be taken.

        Args:
            state (SolverState): Initial state
            ref (Any): Reference data (any pytree)
            max_iters (int): Maximum iterations
            dt_max (float): Largest stepsize
            tol (float): Convergence tolerance on max |v|

        Returns:
            state (SolverState): Converged state
            iters (Array): Number of iterations
        """
        state, ref = jax.lax.stop_gradient((state, ref))
        robot = self.robot
        v, qp_warm_start, info = self.solve(state.q, ref, state.qp_warm_start)
        dt_min = MIN_STEP_RATIO * dt_max

        def cond(carry):
            *_, i, done = carry
            return ~done & (i < max_iters)

        def body(carry):
            q, v, qp_warm_start, cost, dt, i, _ = carry
            q_new = robot.integrate(q, v, dt)
            v_new, qp_warm_start_new, info = self.solve(q_new, ref, qp_warm_start)
            # Accept if the step lowers the cost
            accept = info.cost < cost
            q, v, qp_warm_start, cost = jax.tree.map(
                lambda new, old: jnp.where(accept, new, old),
                (q_new, v_new, qp_warm_start_new, info.cost),
                (q, v, qp_warm_start, cost),
            )
            # If accepted, double dt, otherwise halve it
            dt = jnp.where(accept, jnp.minimum(2.0 * dt, dt_max), 0.5 * dt)
            # Stopping condition on the max |v| or on a minimum dt
            done = (accept & (jnp.max(jnp.abs(v)) <= tol)) | (dt < dt_min)
            return q, v, qp_warm_start, cost, dt, i + 1, done

        carry = (
            state.q,
            v,
            qp_warm_start,
            info.cost,
            jnp.asarray(dt_max, dtype=state.q.dtype),
            0,
            jnp.max(jnp.abs(v)) <= tol,
        )
        q, _, qp_warm_start, _, _, iters, _ = jax.lax.while_loop(cond, body, carry)
        return SolverState(q=q, qp_warm_start=qp_warm_start), iters

class JointLimits(InequalityConstraint):
    """Joint position and velocity limits on the actuated joint velocities

        max(v_min, gain * (q_min - q)) <= v <= min(v_max, gain * (q_max - q))

    Limits default to "robot" to use the robot's own actuated joint position/
    velocity limits, but you can also pass None (no limit) or an array of
    custom limits.

    Currently, limits on floating bases are not implemented.

    Args:
        robot (Robot): The frax robot model
        q_min (ArrayLike | str | None): Lower position limits, shape (num_actuated,).
            Defaults to robot.actuated_joint_lower_limits
        q_max (ArrayLike | str | None): Upper position limits, shape (num_actuated,).
            Defaults to robot.actuated_joint_upper_limits
        v_max (ArrayLike | str | None): Upper velocity limits, shape (num_actuated,).
            Defaults to robot.actuated_joint_max_velocities
        v_min (ArrayLike | str | None): Lower velocity limits, shape (num_actuated,).
            Defaults to -v_max
        gain (float): CBF gain (1/s): how fast the position limits may be approached
        penalty (float): Relaxation penalty
    """

    def __init__(
        self,
        robot: Robot,
        q_min: ArrayLike | Literal["robot"] | None = "robot",
        q_max: ArrayLike | Literal["robot"] | None = "robot",
        v_max: ArrayLike | Literal["robot"] | None = "robot",
        v_min: ArrayLike | Literal["robot"] | None = "robot",
        gain: float = 10.0,
        penalty: float = DEFAULT_PENALTY,
    ):
        assert gain > 0
        self.robot = robot
        n = robot.nv - robot.nv_floating  # Actuated joints
        self.q_min = _resolve_limits(q_min, robot.actuated_joint_lower_limits, n, -1)
        self.q_max = _resolve_limits(q_max, robot.actuated_joint_upper_limits, n, 1)
        self.v_max = _resolve_limits(v_max, robot.actuated_joint_max_velocities, n, 1)
        self.v_min = _resolve_limits(v_min, -self.v_max, n, -1)
        self.upper_idxs = np.flatnonzero(
            np.isfinite(self.q_max) | np.isfinite(self.v_max)
        )
        self.lower_idxs = np.flatnonzero(
            np.isfinite(self.q_min) | np.isfinite(self.v_min)
        )
        I = np.eye(n)
        G = np.vstack([I[self.upper_idxs], -I[self.lower_idxs]])
        self.num_rows = G.shape[0]
        assert self.num_rows > 0
        # Zero columns for the floating base
        self.G = np.hstack([np.zeros((self.num_rows, robot.nv_floating)), G])
        self.gain = gain
        self.penalty = penalty

    def rows(self, q, ref):
        pos = q[self.robot.nq_floating :]
        up, lo = self.upper_idxs, self.lower_idxs
        upper = jnp.minimum(self.v_max[up], self.gain * (self.q_max[up] - pos[up]))
        lower = jnp.maximum(self.v_min[lo], self.gain * (self.q_min[lo] - pos[lo]))
        return self.G, jnp.concatenate([upper, -lower])

        self,

        obstacles: SphereObstacles,
        sphere_idxs: ArrayLike | None = None,
        ref_key: str | None = None,
        max_rows: int | None = None,
        gain: float = 10.0,
        penalty: float = DEFAULT_PENALTY,
        margin: float = 0.0,
    ):
        super().__init__(robot, gain, penalty)
        self.margin = margin
        self.obstacles = SphereObstacles(
            np.asarray(obstacles.centers, dtype=float),
            np.asarray(obstacles.radii, dtype=float),
            np.asarray(obstacles.active, dtype=bool),
        )

            np.arange(len(self.radii))
            if sphere_idxs is None

            "Obstacle count must match the constructor"

        idxs = self.sphere_idxs
        delta = (pos[idxs][None] - obstacles.centers[:, None]).reshape(-1, 3)
        offsets = self.radii[idxs][None] + obstacles.radii[:, None] + self.margin
        offsets = offsets.reshape(-1)
        active = jnp.repeat(obstacles.active, len(idxs))
        dist = jnp.linalg.norm(delta, axis=-1)
        h = jnp.where(active, dist - offsets, _INACTIVE_DISTANCE)
        keep = self._keep(h)
        spheres = jnp.asarray(self.row_spheres)[keep]
        Jv, Jw = self.robot._joint_jacobians(joint_tfs)
        J = self.robot._link_collision_jacobians(joint_tfs, Jv, Jw, spheres)
        normals = delta[keep] / dist[keep, None]
        dh = jnp.einsum("ki,kij->kj", normals, J)
        dh = jnp.where(active[keep, None], dh, 0.0)
        return -dh, self.gain * h[keep]

class Retargeter:
    """Differential IK-based motion retargeter.

    Given a sequence of frames, this class solves for the robot
    configurations q, based on the input DiffIKSolver

    As opposed to just using a DiffIK solver, this adds logic for
    converging to an initial configuration on the first step, and
    processing a whole sequence in a single scan

    Args:
        solver (DiffIKSolver): Diff IK solver
        preprocess (Callable): Input frame -> the solver's reference
        initial_q (Callable): First input frame -> initial guess for q
        converge_max_iters (int): Max iterations to converge onto the first frame
        converge_dt (float | None): Largest convergence stepsize. Defaults to
            1 / solver.max_gain (full Gauss-Newton step)
        converge_tol (float): Convergence tolerance on max |v|
        global_ik_init (GlobalIKSolver | None): If set, solves the first frame by
            global IK before converging, to avoid poor local solutions.
        global_ik_seed (int): Seed for the global IK samples
    """

    def __init__(
        self,
        solver: DiffIKSolver,
        preprocess: Callable[[Any], Any],
        initial_q: Callable[[Any], Array],
        converge_max_iters: int = 300,
        converge_dt: float | None = None,
        converge_tol: float = 1e-3,
        global_ik_init: GlobalIKSolver | None = None,
        global_ik_seed: int = 0,
    ):
        self.solver = solver
        self.preprocess = preprocess
        self.initial_q = initial_q
        self.converge_max_iters = converge_max_iters
        if converge_dt is None:
            assert solver.max_gain is not None, "A term has no gain: pass converge_dt"
            converge_dt = 1.0 / solver.max_gain
        self.converge_dt = converge_dt
        self.converge_tol = converge_tol
        self.global_ik_init = global_ik_init
        self.global_ik_seed = global_ik_seed

    def init(self, frame: Any) -> tuple[SolverState, Array]:
        """Converge onto the first frame

        Returns:
            state (SolverState): Converged state
            iters (Array): Number of iterations
        """
        q, ref = self.initial_q(frame), self.preprocess(frame)
        if self.global_ik_init is not None:
            key = jax.random.key(self.global_ik_seed)
            q = self.global_ik_init.solve(q, ref, key).q
        state, iters = self.solver.converge(
            self.solver.init(q),
            ref,
            self.converge_max_iters,
            self.converge_dt,
            self.converge_tol,
        )
        return state, iters

    def step(
        self, state: SolverState, frame: Any, dt: float
    ) -> tuple[SolverState, SolverInfo]:
        """Take one solver step towards an input frame"""
        state, _, info = self.solver.step(state, self.preprocess(frame), dt)
        return state, info

    @jax.jit
    def retarget_clip(self, frames: Any, dt: float) -> tuple[Array, SolverInfo, Array]:
        """Retarget a full clip of input frames

        Args:
            frames (Any): Input frames, with leading time axis T (any pytree)
            dt (float): Time between frames

        Returns:
            q (Array): Configurations, shape (T, nq)
            infos (SolverInfo): Per-step info, leading dim T - 1
            init_iters (Array): Iterations to converge onto the first frame
        """
        first = jax.tree.map(lambda x: x[0], frames)
        rest = jax.tree.map(lambda x: x[1:], frames)
        init_state, init_iters = self.init(first)

        def scan_step(state, frame):
            state, info = self.step(state, frame, dt)
            return state, (state.q, info)

        _, (q, infos) = jax.lax.scan(scan_step, init_state, rest)
        return jnp.concatenate([init_state.q[None], q]), infos, init_iters

    class OfflineRetargeter:
    """Retargets many clips, spread across devices. Clips are padded to a multiple
    of bucket frames, and compiled shapes are cached across calls

    Args:
        retargeter (Retargeter): Retargeter
        dt (float): Time between frames
        bucket (int): Padded clip lengths are multiples of this
        devices ("all" | "default" | Sequence[jax.Device]): Devices to spread the
            clips across. Defaults to "all"
    """

    def __init__(
        self,
        retargeter: Retargeter,
        dt: float,
        bucket: int = 1024,
        devices: Literal["all", "default"] | Sequence[jax.Device] = "all",
    ):
        self.devices = _resolve_devices(devices)
        self.bucket = bucket
        self.retarget = jax.jit(
            device_vmap(
                lambda clip: retargeter.retarget_clip(clip, dt)[0], self.devices
            )
        )
        self.compiled = {}
        self.compile_time = 0.0

    def __call__(self, clips: Sequence) -> list[np.ndarray]:
        """Retarget clips

        Args:
            clips (Sequence): Clips of input frames, each a pytree with leading
                time axis T. T may differ between clips

        Returns:
            list[np.ndarray]: Configurations, one array per clip, each shape
                (T, nq) with that clip's T
        """
        if not clips:
            return []
        # No more devices than clips, so no device solves a filler clip
        n = min(len(self.devices), len(clips))
        lengths = [len(jax.tree.leaves(clip)[0]) for clip in clips]
        order = sorted(range(len(clips)), key=lambda i: -lengths[i])
        groups = [order[i : i + n] for i in range(0, len(order), n)]
        pending = []
        for group in groups:
            length = -(-lengths[group[0]] // self.bucket) * self.bucket
            # Top up the last group so it shares the compiled shape
            members = [clips[i] for i in group] + [clips[group[0]]] * (n - len(group))
            batch = jax.tree.map(
                lambda *x: np.stack(x),
                *[_pad_with_last_frame(clip, length) for clip in members],
            )
            leaves, structure = jax.tree.flatten(batch)
            key = (structure, tuple(x.shape for x in leaves))
            if key not in self.compiled:
                start = time.perf_counter()
                self.compiled[key] = self.retarget.lower(batch).compile()
                self.compile_time += time.perf_counter() - start
            # Asynchronous dispatch
            pending.append(self.compiled[key](batch))
        qs = [None] * len(clips)
        for group, q in zip(groups, pending):
            q = np.asarray(q)
            for k, i in enumerate(group):
                qs[i] = q[k, : lengths[i]]
        return qs

    class OnlineRetargeter:
    """Stateful retargeting wrapper for online operation.

    Call compile() before any strict real-time operation to
    compile the performance-critical methods.
    """

    def __init__(self, retargeter: Retargeter):
        self.retargeter = retargeter
        self.state = None
        self._init = jax.jit(retargeter.init)
        self._step = jax.jit(retargeter.step)

    def compile(self, example_frame: Any) -> None:
        """Compile the init and step functions ahead of time

        Args:
            example_frame (Any): Example input frame (may be jax.ShapeDtypeStructs)
        """
        state, _ = jax.eval_shape(self._init, example_frame)
        self._init.lower(example_frame).compile()
        self._step.lower(state, example_frame, 0.0).compile()

    def reset(self) -> None:
        self.state = None

    def step(self, frame: Any, dt: float) -> Array:
        if self.state is None:
            self.state, _ = self._init(frame)
        else:
            self.state, _ = self._step(self.state, frame, float(dt))
        return self.state.q

class GlobalIKSolver:
    """Multi-start SQP global IK solver


    Args:
        solver (DiffIKSolver): Diff IK solver
        num_samples (int): Number of starts, including q
        q_min, q_max (ArrayLike | "robot"): Sampling range for the actuated
            joints. Default to "robot", for the robot's joint limits, narrowed
            to those of any JointLimits constraints in the solver
        max_iters (int): Max convergence iterations per start
        dt_max (float | None): Largest convergence stepsize. Defaults to
            1 / solver.max_gain
        tol (float): Convergence tolerance on max |v|
        cost_tol (float | None): Cost tolerance for acceptable solutions. If
            this term is set, then we return the closest sampled solution that is
            within cost_tol of the lowest cost. Defaults to None (select the lowest
            cost regardless of how far the solution is in configuration space)
        devices ("all" | "default" | Sequence[jax.Device]): Devices to parallelize
            the samples across. Defaults to "default" (JAX's default device).
    """

    def __init__(
        self,
        solver: DiffIKSolver,
        num_samples: int = 32,
        q_min: ArrayLike | Literal["robot"] = "robot",
        q_max: ArrayLike | Literal["robot"] = "robot",
        max_iters: int = 200,
        dt_max: float | None = None,
        tol: float = 1e-4,
        cost_tol: float | None = None,
        devices: Literal["all", "default"] | Sequence[jax.Device] = "default",
    ):
        self.solver = solver
        self.num_samples = num_samples

        # Handle joint limits for determining a sampling range
        lower = np.asarray(solver.robot.actuated_joint_lower_limits, dtype=float)
        upper = np.asarray(solver.robot.actuated_joint_upper_limits, dtype=float)
        for c in solver.constraints:
            if isinstance(c, JointLimits):
                lower = np.maximum(lower, c.q_min)
                upper = np.minimum(upper, c.q_max)
        for x in (q_min, q_max):
            if isinstance(x, str):
                assert x == "robot", (
                    f'Sampling limits must be arrays or "robot", got {x!r}'
                )
        self.q_min = lower if isinstance(q_min, str) else np.asarray(q_min, dtype=float)
        self.q_max = upper if isinstance(q_max, str) else np.asarray(q_max, dtype=float)
        assert np.all(np.isfinite(self.q_min)) and np.all(np.isfinite(self.q_max)), (
            "Unbounded sampling range: pass q_min and q_max, or give the solver "
            + "JointLimits"
        )

        self.max_iters = max_iters
        if dt_max is None:
            assert solver.max_gain is not None, "A term has no gain: pass dt_max"
            dt_max = 1.0 / solver.max_gain
        self.dt_max = dt_max
        self.tol = tol
        self.cost_tol = cost_tol
        self.devices = tuple(_resolve_devices(devices))

    def solve(self, q: Array, ref: Any, key: Array) -> GlobalIKResult:
        """Solve global IK

        Args:
            q (Array): Current configuration, shape (nq,)
            ref (Any): Reference data (any pytree)
            key (Array): PRNG key for the samples

        Returns:
            GlobalIKResult: Result
        """
        solver, robot = self.solver, self.solver.robot
        joints = slice(robot.nq_floating, None)

        def converge(q0, ref):
            state, iters = solver.converge(
                solver.init(q0), ref, self.max_iters, self.dt_max, self.tol
            )
            return state.q, iters, solver.cost(state.q, ref)

        def select(qs, costs, iters, sampled):
            if self.cost_tol is None:
                best = jnp.argmin(costs)
            else:
                acceptable = costs <= jnp.min(costs) + self.cost_tol
                distance = jnp.sum((qs[:, joints] - q[joints]) ** 2, axis=1)
                best = jnp.argmin(jnp.where(acceptable, distance, jnp.inf))
            return GlobalIKResult(qs[best], costs[best], qs, costs, iters, sampled)

        def sample():
            q0s = sample_configurations(
                robot, q, key, self.num_samples, self.q_min, self.q_max
            )
            converge_all = device_vmap(converge, self.devices, in_axes=(0, None))
            qs, iters, costs = converge_all(q0s, ref)
            return select(qs, costs, iters, jnp.array(True))

        if self.cost_tol is None:
            return sample()
        q_local, iters_local, cost_local = converge(q, ref)

        def keep_local():
            tile = lambda x: jnp.broadcast_to(x, (self.num_samples,) + jnp.shape(x))
            return select(
                tile(q_local), tile(cost_local), tile(iters_local), jnp.array(False)
            )

        return jax.lax.cond(cost_local <= self.cost_tol, keep_local, sample)

        Array: Sampled configurations, shape (num_samples, nq)

        assert q_max == "robot", f'q_max must be an array or "robot", got {q_max!r}'

        assert np.all(np.isfinite(q_max)), "Unbounded joint limits: pass q_max"

        key,

        q[: robot.nq_floating], (num_samples - 1, robot.nq_floating)
    )
    return jnp.concatenate([q[None], jnp.concatenate([base, joints], axis=1)])

    def load_bvh(
    filepath: str | Path, to_world: ArrayLike | None = None, scale: float = 0.01
) -> tuple[np.ndarray, list[str]]:
    """Global bone poses from a BVH file

    Args:
        filepath (str | Path): BVH file
        to_world (ArrayLike | None): Rotation from the BVH frame to the world,
            shape (3, 3). Defaults to identity
        scale (float): BVH length unit (m). Defaults to cm

    Returns:
        poses (np.ndarray): Poses [position, WXYZ quaternion], shape (T, B, 7)
        names (list[str]): The B bone names

    Raises:
        ValueError: If any frame has NaNs
    """
    names, parents, local_pos, local_rot = _parse_bvh(filepath)
    T, B = local_pos.shape[:2]
    rot = np.zeros((T, B, 3, 3))
    pos = np.zeros((T, B, 3))
    for i, parent in enumerate(parents):  # Parents precede their children
        if parent < 0:
            rot[:, i], pos[:, i] = local_rot[:, i], local_pos[:, i]
        else:
            rot[:, i] = rot[:, parent] @ local_rot[:, i]
            pos[:, i] = pos[:, parent] + np.einsum(
                "tij,tj->ti", rot[:, parent], local_pos[:, i]
            )
    to_world = np.eye(3) if to_world is None else np.asarray(to_world, dtype=float)
    pos = scale * pos @ to_world.T
    quat = Rotation.from_matrix((to_world @ rot).reshape(-1, 3, 3))
    quat = quat.as_quat(scalar_first=True).reshape(T, B, 4)
    return np.concatenate([pos, quat], axis=-1), names

    def smooth_poses(
    poses: ArrayLike, dt: float, sigma_pos: float = 0.0, sigma_rot: float = 0.0
) -> np.ndarray:
    """Zero-phase Gaussian smoothing of a clip of poses along time,
    for offline retargeting

    Args:
        poses (ArrayLike): Poses [position, WXYZ quaternion], shape (T, ..., 7)
        dt (float): Time between poses
        sigma_pos (float): Position kernel standard deviation (s). 0 to not smooth
        sigma_rot (float): Rotation kernel standard deviation (s). 0 to not smooth

    Returns:
        np.ndarray: Smoothed poses, shape (T, ..., 7)
    """
    poses = np.array(poses, dtype=float)
    if sigma_pos > 0:
        poses[..., :3] = gaussian_filter1d(
            poses[..., :3], sigma_pos / dt, axis=0, mode="nearest"
        )
    if sigma_rot > 0:
        quat = poses[..., 3:]
        # Sign-align, so componentwise filtering approximates the geodesic mean
        flips = np.sum(quat[1:] * quat[:-1], axis=-1) < 0
        signs = np.cumprod(np.where(flips, -1.0, 1.0), axis=0)
        quat[1:] *= signs[..., None]
        quat = gaussian_filter1d(quat, sigma_rot / dt, axis=0, mode="nearest")
        poses[..., 3:] = quat / np.linalg.norm(quat, axis=-1, keepdims=True)
    return poses

    def resample_poses(poses: ArrayLike, fps_in: float, fps_out: float) -> np.ndarray:
    """Resample a clip of poses to a new frame rate

    Args:
        poses (ArrayLike): Poses [position, WXYZ quaternion], shape (T, ..., 7)
        fps_in (float): Frame rate of poses
        fps_out (float): Frame rate to resample to

    Returns:
        np.ndarray: Resampled poses, shape (T_out, ..., 7), with
            T_out = floor((T - 1) * fps_out / fps_in) + 1
    """
    poses = np.asarray(poses, dtype=float)
    T = len(poses)
    assert T >= 2, "Need at least two poses to resample"
    flat = poses.reshape(T, -1, 7)
    n = flat.shape[1]

    T_out = int(np.floor((T - 1) * fps_out / fps_in + 1e-9)) + 1
    frame_times = np.arange(T_out) * fps_in / fps_out  # In input frames
    i = np.minimum(frame_times.astype(int), T - 2)
    alpha = frame_times - i
    before, after = flat[i], flat[i + 1]

    a = alpha[:, None, None]
    pos = (1 - a) * before[..., :3] + a * after[..., :3]

    rot_before = Rotation.from_quat(before[..., 3:].reshape(-1, 4), scalar_first=True)
    rot_after = Rotation.from_quat(after[..., 3:].reshape(-1, 4), scalar_first=True)
    relative = (rot_before.inv() * rot_after).as_rotvec()
    a = np.repeat(alpha, n)[:, None]
    rot = rot_before * Rotation.from_rotvec(a * relative)
    quat = rot.as_quat(scalar_first=True).reshape(T_out, n, 4)

    out = np.concatenate([pos, quat], axis=-1)
    return out.reshape(T_out, *poses.shape[1:])
